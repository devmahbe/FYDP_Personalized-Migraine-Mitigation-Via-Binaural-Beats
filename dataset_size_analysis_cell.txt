# Visualize dataset size vs accuracy expectations
import matplotlib.pyplot as plt

fig, axes = plt.subplots(1, 2, figsize=(15, 5))

# Left: Literature comparison
ax1 = axes[0]
studies = ['Our Study\n(n=31)', 'Small Study\n(n=50)', 'Medium Study\n(n=100)', 
           'Large Study\n(n=150)', 'SOTA\n(n=200+)']
accuracies = [0.60, 0.72, 0.80, 0.893, 0.92]
sample_sizes = [31, 50, 100, 150, 200]

colors_vis = ['#e74c3c' if n <= 50 else '#f39c12' if n <= 100 else '#2ecc71' 
              for n in sample_sizes]

bars = ax1.bar(studies, [a*100 for a in accuracies], color=colors_vis, 
               edgecolor='black', alpha=0.8)
ax1.set_ylabel('Accuracy (%)', fontweight='bold', fontsize=12)
ax1.set_title('EEG Migraine Classification: Dataset Size vs Accuracy', 
              fontweight='bold', fontsize=13)

            label='Random Guessing (3 classes)')

            label='Acceptable (75%)')
ax1.axhline(y=85, color='green', linestyle='--', linewidth=2, alpha=0.5, 
            label='State-of-the-art (85%+)')
ax1.set_ylim([0, 100])
ax1.legend(loc='lower right', fontsize=9)
ax1.grid(alpha=0.3, axis='y')
